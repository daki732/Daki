-- ====== FUNCTIONS ======
local function GetRoot(p)
    if not p then return nil end
    local c = p.Character
    if not c then return nil end
    return c:FindFirstChild("HumanoidRootPart")
end

local function UpdateCache()
    local c = LocalPlayer.Character
    State.MyRoot = c and c:FindFirstChild("HumanoidRootPart")
end

local function HasWeapon(p)
    if not p or not p.Character then return false end
    for _, item in ipairs(p.Character:GetChildren()) do
        if item:IsA("Tool") then
            local n = item.Name:lower()
            if n:find("knife") or n:find("blade") or n:find("dagger") or n:find("sword") then
                return true
            end
        end
    end
    return false
end

local function IsMurderer(p)
    if not p then return false end
    if p:GetAttribute("Murderer") == true or p:GetAttribute("isMurderer") == true then return true end
    return HasWeapon(p)
end

local function FindBestTarget()
    local now = os_clock()
    if State.Target and State.Target.Parent == Players then
        local c = State.Target.Character
        local hum = c and c:FindFirstChild("Humanoid")
        if hum and hum.Health > 0 and IsMurderer(State.Target) and (now - State.LastCheck < 0.8) then
            return State.Target
        end
    end
    State.LastCheck = now
    local bestTarget = nil
    local bestDist = math.huge
    for _, pl in ipairs(Players:GetPlayers()) do
        if pl ~= LocalPlayer and IsMurderer(pl) then
            local root = GetRoot(pl)
            local hum = pl.Character and pl.Character:FindFirstChild("Humanoid")
            if root and hum and hum.Health > 0 then
                local dist = (root.Position - State.MyRoot.Position).Magnitude
                if dist < bestDist and dist >= State.Settings.minDistance and dist <= State.Settings.maxDistance then
                    bestDist = dist
                    bestTarget = pl
                end
            end
        end
    end
    State.Target = bestTarget
    return bestTarget
end

local function FastProcessTarget(root)
    if not root then return nil, nil end
    local pos = root.Position
    local vel = root.AssemblyLinearVelocity
    local ph, vh = State.TargetPosHistory, State.TargetVelHistory

    -- Apply moving average for velocity smoothing
    table_insert(ph, pos)
    table_insert(vh, vel)
    if #ph > MAX_HISTORY then table_remove(ph, 1) end
    if #vh > MAX_HISTORY then table_remove(vh, 1) end

    local avgP, avgV = Vector3_new(0, 0, 0), Vector3_new(0, 0, 0)
    for i = 1, #ph do
        avgP = avgP + ph[i]
        avgV = avgV + vh[i]
    end
    avgP = avgP / #ph
    avgV = avgV / #vh

    -- Anti-strafe logic with adjusted thresholds
    if State.Settings.antiStrafe and avgV.Magnitude > 1.8 then
        local dir = Vector3_new(avgV.X, 0, avgV.Z).Unit
        if dir:Dot(State.LastDirection) < 0.2 then
            State.AntiStrafeFactor = 0.65
        else
            State.AntiStrafeFactor = math_clamp(State.AntiStrafeFactor + 0.05, 0.65, 1.2) -- Increased max to 1.2
        end
        State.LastDirection = dir
    else
        State.AntiStrafeFactor = 1.0
    end

    return avgP, avgV
end

local function ApplyGPL(sim, interval, x, y, z, h, v)
    local last = State.LastApplied
    if math_abs(last.H - h) < 2 and math_abs(last.V - v) < 2 and
        math_abs(last.Sim - sim) < 2 and math_abs(last.Int - interval) < 2 and
        last.X == x and last.Y == y and last.Z == z then
        return
    end
    last.H, last.V, last.Sim, last.Int, last.X, last.Y, last.Z = h, v, sim, interval, x, y, z
    if gpl_preset[4] then gpl_preset[4](sim) end
    if gpl_preset[5] then gpl_preset[5](interval) end
    if gpl_preset[6] then gpl_preset[6](x) end
    if gpl_preset[7] then gpl_preset[7](y) end
    if gpl_preset[8] then gpl_preset[8](z) end
    if gpl_preset[9] then gpl_preset[9](h) end
    if gpl_preset[10] then gpl_preset[10](v) end
end

local function InitBase()
    if not internal_shared["RevertSettings_PrioritizeYourPing"] and gpl_preset[1] then gpl_preset[1]() end
    if not internal_shared["RevertSettings_PredictJump"] and gpl_preset[2] then gpl_preset[2]() end
    if not internal_shared["RevertSettings_PredictLag"] and gpl_preset[3] then gpl_preset[3]() end
end

local function Optimize()
    if os_clock() - State.LastGC > 60 then -- Increased from 40 to 60
        State.LastGC = os_clock()
        collectgarbage("step", 200) -- Increased from 100 to 200
    end
end

-- ====== MAIN LOOP (OPTIMIZED) ======
RunService.Heartbeat:Connect(function()
    if not State.Enabled then return end
    Optimize()

    if not State.MyRoot or not State.MyRoot.Parent then
        UpdateCache()
        if not State.MyRoot then return end
    end

    local target = FindBestTarget()
    if not target then return end

    local mRoot = GetRoot(target)
    if not mRoot then return end

    local myPos, mPos = State.MyRoot.Position, mRoot.Position
    local dist = (mPos - myPos).Magnitude

    if dist < State.Settings.minDistance or dist > State.Settings.maxDistance then return end

    local smoothPos, smoothVel = FastProcessTarget(mRoot)
    if not smoothPos then return end

    local ping = LocalPlayer:GetNetworkPing() * 1000
    if ping <= 0 then ping = 80 end -- Increased default ping from 60 to 80

    local horizSpeed = Vector3_new(smoothVel.X, 0, smoothVel.Z).Magnitude
    local mode = MODES[State.CurrentMode] or MODES.PRO

    -- Adaptive mode logic with refined thresholds
    if State.CurrentMode == "ADAPTIVE" then
        if dist < 20 then
            mode = MODES.ADAPTIVE.close
        elseif dist > 150 then
            mode = MODES.ADAPTIVE.long
        else
            mode = MODES.ADAPTIVE.mid
        end
    end

    local antiStrafeMult = State.AntiStrafeFactor
    local hLead = math_clamp(
        math_floor(mode.h_base + (ping * mode.h_ping) + (horizSpeed * mode.h_speed * antiStrafeMult)),
        100, 500 -- Adjusted clamping range
    )
    local vLead = math_clamp(
        math_floor(mode.v_base + (ping * mode.v_ping) + (dist * mode.v_dist)),
        80, 420
    )

    -- Enhanced jump prediction
    if State.Settings.predictJump then
        if smoothVel.Y > 2.5 then -- Adjusted from 3 to 2.5
            vLead = vLead + 35 -- Increased from 30 to 35
        elseif smoothVel.Y < -8 then
            vLead = vLead - 25 -- Increased from 20 to 25
        end
    end

    -- Dynamic simulation and interval
    local sim = math_clamp(
        math_floor(mode.sim_base + (horizSpeed * mode.sim_speed)),
        18, 120
    )
    local interval = math_clamp(
        math_floor(mode.int_base + (horizSpeed * mode.int_speed)),
        8, 100
    )

    -- Dynamic offsets based on distance
    local offX, offY, offZ = mode.offX, mode.offY, mode.offZ
    if dist < 20 then
        offX, offZ = -3, 2
        offY = offY + 1 -- Increased for better headshot alignment
    elseif dist > 150 then
        offX, offZ = 1, -1
    end

    ApplyGPL(sim, interval, offX, offY, offZ, hLead, vLead)
end)
